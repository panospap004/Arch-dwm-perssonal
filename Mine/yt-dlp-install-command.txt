and to check if all the songs where installed in the correct bitrate run this in bash

cd ~/Music/new-downloads && for file in *.mp3; do
  bitrate=$(ffprobe -v error -select_streams a:0 -show_entries stream=bit_rate -of csv=p=0 "$file")
  echo "${file%.mp3} , $((bitrate / 1000)) kbps"
done
